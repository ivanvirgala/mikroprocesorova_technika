\TUKEsection{Praktické zapojenia}{LED, tlačidlo a interný pull-up rezistor}

\begin{frame}[fragile]{Príklad: nastavenie výstupu PB5}
\begin{lstlisting}[style=avrstyle]
#include <avr/io.h>

int main(void)
{
    DDRB |= (1 << DDB5);      // PB5 ako vystup
    PORTB |= (1 << PORTB5);   // logicka 1 na PB5

    while (1)
    {
    }
}
\end{lstlisting}
\end{frame}
