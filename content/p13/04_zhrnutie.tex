\TUKEsection{Zhrnutie predmetu}{Od registrov k mechatronickej aplikácii}

\begin{frame}{Zhrnutie}
\begin{tukebox}{Kľúčové body}
\begin{itemize}
    \item Každá periféria sa ovláda cez konkrétne registre.
    \item Datasheet je primárny zdroj informácií.
    \item Arduino UNO R3 používame ako hardvérovú platformu, nie ako softvérový framework.
    \item Overenie programu má byť podložené meraním alebo pozorovaním signálu.
\end{itemize}
\end{tukebox}
\end{frame}

\begin{frame}{Otázky na kontrolu}
\begin{enumerate}
    \item Ktoré registre alebo bity sú pre túto tému kľúčové?
    \item Aký je rozdiel medzi konfiguráciou a čítaním stavu?
    \item Ako by ste overili správnu funkciu na reálnej doske?
\end{enumerate}
\end{frame}
